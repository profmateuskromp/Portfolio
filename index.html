// Rotina automática de verificação de umidade e rega
void gerenciarOrquidario() {
  int leituraUmidade = analogRead(sensorPin);
  float porcentagemUmidade = map(leituraUmidade, 1023, 200, 0, 100);

  if (porcentagemUmidade < 60.0) {
    digitalWrite(valvulaAguaPin, HIGH); // Abre os aspersores
    delay(5000);                        // Rega por 5 segundos
    digitalWrite(valvulaAguaPin, LOW);  // Fecha
  }
}
